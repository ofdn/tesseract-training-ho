𑢾𑣉𑣗𑣈𑣓 𑣌𑣉𑣈𑣕𑣈 𑣚𑣁𑣓𑣁𑣄 𑣙𑣉𑣉 𑣌𑣁𑣎𑣂 𑣌𑣁𑣎𑣂𑣂 𑣕𑣁𑣓 𑣌𑣉𑣁𑣄 𑣒𑣁𑣈𑣑 𑣔𑣉 𑢬𑣉𑣚𑣙𑣁𑣓 𑣜𑣈𑣁𑣄
𑢵𑣃𑣜𑣞𑣁𑣄 𑢾𑣂𑣍𑣗𑣃𑣆 𑣕𑣁𑣓𑣁 𑣉𑣌𑣉𑣓𑣁 𑣔𑣉 𑣎𑣁𑣜𑣌𑣁𑣓𑣑 𑣜𑣈 𑣖𑣈𑣓𑣁𑣄𑣁𑣁. 𑣈𑣓𑣁 𑣚𑣉𑣄 𑣕𑣈

𑣓𑣈𑣚𑣈 𑣜𑣈 𑢫𑣉𑣒𑣁 𑣎𑣁𑣜𑣌𑣁𑣓𑣑 𑣜𑣈 . 𑢹𑣉𑣉 𑣎𑣁𑣋𑣁𑣜 𑣕𑣁𑣓𑣌𑣉 𑣖𑣈𑣓𑣁𑣄 𑣌𑣉. 𑢨𑣓 𑣚𑣈𑣌𑣁 𑣋𑣈
𑢩𑣑𑣂𑣝𑣁 𑣜𑣁𑣆𑣎 𑣜𑣈 𑣎𑣂𑣚𑣁 𑣜𑣈 𑢹𑣉𑣉 𑣎𑣁𑣋𑣁𑣜 𑣕𑣁𑣓 𑣙𑣉𑣛𑣌𑣉 𑣖𑣈𑣓𑣁𑣄 𑣌𑣉𑣁 𑣁𑣓𑣑𑣉𑣄 𑣞𑣉𑣗𑣈𑣓
