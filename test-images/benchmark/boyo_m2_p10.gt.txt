𑢹𑣕𑣃 𑣙𑣕𑣃 𑢴𑣈𑣝𑣁𑣇𑣚𑣂 𑣜𑣂𑣆𑣎𑣂 𑣎𑣌𑣈 𑣋𑣉𑣒𑣁 𑣖𑣃𑣚𑣂
𑢮𑣃𑣖𑣃𑣜 𑣎𑣃𑣖𑣃𑣜 𑣔𑣃𑣖𑣃𑣜 𑣚𑣈𑣌𑣁 𑣟𑣂𑣜𑣖𑣁 𑣜𑣂𑣖𑣂𑣚 𑣚𑣈𑣌𑣁
𑢮𑣊𑣋𑣁𑣗𑣈𑣓𑣌𑣁 𑣎𑣚𑣁𑣊𑣋𑣂𑣓𑣌𑣁 𑣁𑣌𑣁𑣛𑣁 𑣔𑣃𑣔𑣃𑣋𑣜
𑢷𑣃𑣜𑣃𑣓𑣁 𑣗𑣂𑣕𑣜 𑣌𑣉𑣏𑣁𑣓𑣁 𑣌𑣉𑣕𑣈 𑣜𑣂𑣆𑣎𑣂 𑣘𑣜𑣉𑣖 𑣔𑣂𑣝𑣃𑣖 𑣌𑣉𑣕𑣈
𑢳𑣂𑣕𑣂𑣜𑣈𑣓𑣁 𑣏𑣉𑣚𑣀 𑣈𑣓𑣁 𑣖𑣋𑣈 𑣔𑣃𑣔𑣃𑣋𑣜
𑢮𑣕𑣂 𑣔𑣙𑣉𑣜𑣉𑣖 𑣟𑣉𑣖𑣎 𑣟𑣋𑣂𑣆 𑣙𑣂𑣘𑣂𑣜𑣃𑣖𑣕𑣓 𑣘𑣉𑣜𑣉𑣗 𑣘𑣓𑣂𑣆
𑢶𑣂𑣟𑣁𑣕𑣈 𑣖𑣃𑣟𑣂𑣀 𑣖𑣓𑣁𑣕𑣂𑣀𑣜𑣈 𑣔𑣂𑣝𑣃𑣖 𑣔𑣃𑣔𑣃𑣋𑣜
𑢾𑣜𑣂 𑣟𑣜𑣂 𑣔𑣈𑣝𑣁𑣇𑣚𑣂 𑣓𑣂𑣕𑣂𑣜𑣁 𑣌𑣓𑣁𑣗𑣃 𑣋𑣉𑣒𑣁𑣖𑣃𑣚𑣂
𑢾𑣂𑣀𑣋𑣂 𑣏𑣓𑣑𑣃𑣄 𑣕𑣃𑣜 𑣎𑣉𑣌𑣁 𑣁𑣌𑣁𑣛𑣁 𑣔𑣃𑣔𑣃𑣋𑣜
𑢷𑣊𑣚𑣁𑣔𑣂𑣞𑣃𑣖 𑣌𑣃𑣀𑣔𑣃-𑣕𑣃𑣜 𑢨𑣈𑣞𑣂𑣅𑣈 𑣜𑣈 𑣖𑣈𑣓𑣄 𑣕𑣓 𑣖𑣂𑣅𑣑 𑣔𑣂𑣞𑣃𑣖 𑣕𑣓𑣁. 𑢳𑣈𑣓𑣁 𑣜𑣈𑣅𑣄
𑣁𑣜𑣞𑣆 𑢱𑣁𑣌𑣁
''𑢷𑣜𑣖𑣂'' 𑣔𑣉 𑢹𑣜𑣕𑣙𑣞𑣁 𑣜𑣈𑣓 𑣚𑣁𑣄𑣋𑣈𑣓 𑣏𑣂𑣕𑣂𑣅𑣜 𑣌𑣉 𑣉𑣄 𑢬𑣈𑣖𑣈𑣜-𑢵𑣂𑣗𑣈𑣑 𑣏𑣂𑣕𑣂𑣅𑣜 𑣌𑣉
𑣜𑣈𑣅𑣄 𑣞𑣂𑣔𑣈𑣊𑣋𑣁 𑣏𑣂𑣕𑣂𑣅𑣜 𑣕𑣓𑣁. 𑢳𑣈𑣓 𑣏𑣂𑣕𑣂𑣅𑣜 𑢹𑣜𑣕𑣙𑣞𑣁 𑣜𑣈 𑣞𑣞 -𑣥𑣠𑣠 𑣓𑣉𑣄 𑣜𑣈 𑣘𑣅𑣁𑣜
𑣎𑣓𑣁. 𑢳𑣈𑣓 𑣏𑣂𑣕𑣂𑣅𑣜 𑣔𑣉 𑢡𑣜𑣖𑣈𑣈 𑣌𑣁𑣜𑣈 𑢸𑣂𑣓𑣂𑣞𑣈𑣓 𑣏𑣂𑣕𑣂𑣅𑣜 𑣈𑣕𑣈 𑣗𑣆 𑣉𑣄𑣚𑣁𑣌𑣓𑣁 𑣖𑣈𑣓𑣕𑣈
𑣖𑣂𑣅𑣑 𑣇𑣗𑣈𑣜 𑣖𑣈𑣓𑣄. 𑢩𑣄 𑣖𑣂𑣅𑣑 𑣇𑣗𑣈𑣜 𑣚𑣈𑣌𑣁 𑣕𑣈 𑣓𑣈𑣓 𑣏𑣂𑣕𑣂𑣅𑣜 𑣔𑣉 𑣞𑣂𑣀𑣔𑣃 𑢠-𑣕𑣅𑣑 𑣜𑣈
𑣞𑣞 -𑣢𑣠𑣠𑣠 𑣆𑣖𑣂𑣕𑣊 𑣘𑣅𑣁𑣜𑣉𑣄 𑣕𑣆𑣌𑣈𑣓 𑢹𑣛𑣁𑣘𑣁 𑣏𑣂𑣕𑣂𑣅𑣜 𑣈𑣕𑣈 𑣗𑣆 𑣉𑣄𑣚𑣁𑣌𑣓𑣁. 𑢳𑣈𑣓
𑣏𑣂𑣕𑣂𑣅𑣜 𑣕𑣈 𑣉𑣚𑣁𑣌𑣓 𑣞𑣘𑣜𑣂 𑣉𑣓𑣉𑣚 𑣌𑣉 𑣔𑣉 𑣋𑣉𑣞𑣈𑣓 𑢡𑣞𑣉𑣉 (𑣞𑣞 𑣢𑣧𑣠 𑣞𑣞 𑣢𑣣𑣢) 𑣁𑣄 𑣌𑣉
𑣕𑣓𑣁. 𑢳𑣈𑣓 𑣏𑣂𑣕𑣂𑣅𑣜 𑢾𑣊𑣌𑣂𑣜, 𑢸𑣜𑣌𑣂𑣜 𑣉𑣄 𑣈𑣞𑣊𑣋𑣂 𑣈𑣒𑣄 𑣎𑣈𑣋𑣜 𑣌𑣉 𑣉𑣚 𑣜𑣈 𑣘𑣅𑣁𑣜𑣁𑣌𑣓𑣁.
𑢳𑣈𑣓 𑣏𑣂𑣕𑣂𑣅𑣜 𑣚𑣈𑣊𑣋𑣁 𑣈𑣕𑣈 𑣈𑣕𑣉𑣖 𑣘𑣁 𑣕𑣈 𑣉𑣚-𑣆𑣑𑣉𑣄𑣁
𑢸𑣂𑣓𑣙𑣞𑣁 𑣏𑣗𑣜 𑢥𑣃𑣃𑣜𑣉𑣗 𑣜𑣈 𑣖𑣈𑣓𑣄 𑣕𑣓 𑣖𑣂𑣅𑣑 𑣔𑣂𑣞𑣃𑣖 𑣕𑣓𑣁. 𑢳𑣈𑣓𑣁 𑣜𑣈𑣅𑣄 𑣁𑣜𑣞𑣆
𑢹𑣈𑣚𑣞𑣂𑣊𑣌𑣂
𑢺𑣉𑣞𑣈𑣜 𑣔𑣉 𑣓𑣈𑣚-𑣗𑣓𑣁𑣆 𑣕𑣋𑣈𑣓 𑣔𑣆𑣅𑣉𑣄 𑣖𑣂𑣚𑣈𑣌𑣁𑣓 𑣎𑣃𑣓𑣂𑣑 𑣕𑣓𑣁. 𑢺𑣉𑣞𑣈𑣜 𑣕𑣈 𑣎𑣃𑣓𑣂𑣑 𑣗𑣆
𑣚𑣗𑣁𑣞𑣜 𑣋𑣈 𑣕𑣈 𑣈𑣞𑣁𑣊𑣋𑣂 𑣎𑣃𑣓𑣂𑣑 𑣌𑣉 𑣓𑣈𑣓𑣁 𑣕𑣈 𑣗𑣆𑣅𑣉𑣄𑣁. 𑢺𑣉𑣞𑣈𑣜 𑣜𑣈 𑣈𑣞𑣃 𑣚𑣈𑣌𑣁
𑣖𑣈𑣓𑣄. 𑢩𑣌𑣉𑣌𑣉𑣓 𑣔𑣉 𑣖𑣂𑣞𑣈 𑣗𑣆 𑣌𑣈𑣈𑣑 𑣉𑣀𑣑𑣉𑣄 𑣌𑣁 𑣞𑣈𑣜𑣉𑣄𑣁, 𑣉𑣌𑣉𑣌𑣉𑣓 𑣔𑣉 𑣚𑣉𑣚𑣉 𑣌𑣈𑣈
𑣜𑣈 𑣚𑣈𑣗𑣈𑣄 𑣓𑣁. 𑢺𑣁𑣄𑣕𑣜 𑣚𑣉𑣞𑣈𑣜 𑣔𑣉 𑣙𑣉𑣛𑣉 𑣕𑣈 𑣗𑣆𑣅𑣉𑣄𑣁, 𑣙𑣞𑣁𑣗𑣂𑣜 𑣜𑣈 𑣌𑣁 𑣓𑣖𑣉𑣄𑣁.
𑢺𑣉𑣞𑣈𑣜 𑣗𑣆 𑣜𑣈𑣅𑣄 𑣏𑣂𑣚𑣂𑣋𑣈 𑣕𑣈𑣜𑣋𑣈𑣍 𑣋𑣈𑣅𑣁. 𑢺𑣉𑣞𑣈𑣜 𑣖𑣈𑣓𑣉𑣄 𑣚𑣁𑣄𑣕𑣜 𑣎𑣃𑣓𑣂𑣑 𑣌𑣉 𑣔𑣉
𑢸𑣉𑣚𑣂𑣖𑣈𑣜 𑣌𑣉 𑣕𑣓𑣁. 𑢸𑣉𑣚𑣂𑣖𑣈𑣜 𑣌𑣉 𑣔𑣉 𑣎𑣂𑣚𑣂𑣊 𑣕𑣈 𑣞𑣘𑣁𑣗 𑣁𑣌𑣁𑣓 𑣎𑣄𑣎𑣊 𑣞𑣂𑣊𑣌𑣂𑣛𑣂 𑣌𑣉
