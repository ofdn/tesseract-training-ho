𑣕𑣓𑣁
𑢹𑣞𑣁𑣗𑣂𑣜 𑢸𑣉𑣚𑣂𑣖𑣈𑣜 𑣕𑣈 𑣈𑣞𑣃 𑣁𑣅𑣁𑣜 𑣋𑣈 𑣙𑣉𑣛𑣉 𑣌𑣉 𑣚𑣉𑣞𑣈𑣜 𑣌𑣉 𑣗𑣆-𑣞𑣗 𑣚𑣈𑣌𑣁𑣑𑣁.
𑢡𑣚𑣈𑣑𑣞𑣈𑣀𑣑𑣈𑣜 𑢸𑣁𑣜𑣌𑣞, 𑢦𑣊𑣋𑣂𑣚𑣙𑣞𑣁 𑣜𑣈𑣓 𑣗𑣆-𑣓𑣖𑣋𑣜, 𑣡𑣨𑣥𑣥 𑣜𑣈 𑣕𑣂𑣞𑣂𑣊𑣄 𑣚𑣉𑣞𑣈𑣜
𑣚𑣈𑣌𑣁𑣓 𑣎𑣃𑣓𑣂𑣑𑣎 𑣗𑣆 𑣌𑣈𑣑𑣁, 𑣈𑣓𑣁 𑣌𑣈𑣛𑣈𑣄 𑣋𑣈 𑣕𑣆𑣌𑣈𑣓𑣁 𑣖𑣈𑣓𑣔𑣉 𑣚𑣒𑣃𑣌𑣈𑣖 𑣋𑣈 𑣉𑣉
𑣕𑣆𑣌𑣈𑣓𑣁. 𑣖𑣈𑣓𑣕𑣈𑣎 𑣓𑣃𑣕𑣃𑣖 𑣚𑣈𑣑𑣁, 𑣈𑣓𑣁 𑣋𑣈 𑣓𑣁𑣄 𑣖𑣈𑣓𑣉𑣄𑣁
𑢸𑣘𑣁𑣜𑣂 𑣎𑣓 𑣚𑣉𑣞𑣈𑣜 𑣔𑣉 𑣎𑣁 𑣜𑣈 𑣋𑣈 𑣈𑣛𑣄 𑣓𑣈. 𑢩𑣌𑣉𑣌𑣉𑣓 𑣚𑣉𑣞𑣜 𑣌𑣉 𑣔𑣉 𑣚𑣉𑣉 𑣏𑣂 𑣚𑣉𑣚𑣉
𑣌𑣈𑣈 𑣜𑣈 𑣗𑣂𑣞𑣂 𑣖𑣉𑣄 𑣉𑣄𑣚𑣁. 𑢵𑣜𑣁 𑣚𑣉𑣞𑣈𑣜 𑣔𑣉 𑣁𑣚𑣁𑣌𑣁 𑣈𑣛𑣄 𑣞𑣖𑣁𑣄𑣉 𑣖𑣈𑣓𑣕𑣈
𑣗𑣂𑣅𑣃𑣜-𑣞𑣈𑣓-𑣗𑣆𑣅𑣉𑣄𑣁
𑢹𑣉𑣉 𑢮𑣋𑣁𑣜 𑢼𑣂𑣞𑣉𑣉𑣜𑣞 𑣌𑣉
𑢹𑣉𑣉 𑣎𑣋𑣁𑣜 𑣔𑣉 𑢮𑣁𑣜𑣌𑣀𑣑, 𑢩𑣑𑣂𑣞𑣁, 𑢹𑣞𑣃𑣜 𑣞𑣄 𑢷𑣊𑣋𑣁𑣚 𑣜𑣈 𑣕𑣆𑣓 𑣈𑣈𑣀𑣋𑣈𑣚 𑢹𑣉𑣉 𑣎𑣕𑣂 𑣌𑣉
𑣁𑣄 𑣚𑣈𑣄𑣌𑣎𑣂 𑣕𑣓𑣁. 𑢡𑣅𑣁𑣜 𑣔𑣉 𑢹𑣉𑣉 𑣎𑣋𑣁𑣜 𑢦𑣊𑣚𑣂𑣞, 𑢴𑣈𑣉𑣓𑣁𑣋𑣜𑣂, 𑢩𑣑𑣂𑣅𑣁, 𑢷𑣁𑣊𑣚𑣁
𑣈𑣖𑣁𑣓 𑣌𑣉 𑣕𑣈 𑣉𑣚𑣉𑣄 𑣕𑣆𑣌𑣈𑣓𑣁, 𑣓𑣁𑣄 𑣔𑣉 𑢹𑣗𑣁𑣜𑣊 𑢯𑣂𑣕𑣂 𑣕𑣈 𑣋𑣈 𑣉𑣚𑣉𑣄 𑣕𑣓𑣁. 𑢹𑣗𑣁𑣜𑣊
𑢯𑣂𑣕𑣂 𑣔𑣉 𑢹𑣉𑣉 𑣙𑣃𑣔𑣁 𑣜𑣈𑣓 𑣖𑣂𑣈𑣑 𑣞𑣈𑣀𑣅𑣁𑣓 𑣙𑣉𑣉 𑢺𑣌𑣉 𑢷𑣉𑣔𑣜𑣈 𑣗𑣆 𑣚𑣈𑣑𑣁. 𑢥𑣃𑣓𑣈𑣞𑣌𑣉
𑣔𑣉 𑣓𑣈𑣓 𑣎𑣋𑣁𑣜 𑣉𑣀𑣑𑣉𑣄 𑣏𑣂𑣖𑣂𑣓𑣈𑣊 𑣞𑣂𑣜𑣖𑣈 𑣜𑣈 𑣁𑣑𑣉𑣄𑣁 𑣖𑣈𑣓𑣕𑣈𑣄 𑣖𑣓𑣁𑣕𑣂𑣊 𑣕𑣑𑣁
𑢳𑣁𑣄 𑢹𑣉𑣉 𑣎𑣋𑣁𑣜 𑢹𑣗𑣁𑣜𑣊 𑢯𑣂𑣕𑣂 𑣕𑣈 𑢿𑣂𑣌𑣂𑣘𑣂𑣂𑣑𑣂𑣅𑣁 𑣗𑣆𑣅𑣉𑣄 𑣕𑣓𑣁, 𑣓𑣁𑣓𑣁 𑣖𑣈𑣓𑣔𑣉 𑣁𑣇𑣜𑣂
𑣗𑣆 𑣞𑣈𑣌𑣁𑣄𑣉𑣁, 𑣘𑣆𑣒𑣂 𑣞𑣈𑣞𑣈𑣓 𑣕𑣓𑣁. 𑢹𑣉𑣉 𑣕𑣈 𑢿𑣂𑣌𑣂𑣘𑣂𑣂𑣑𑣂𑣅𑣁 𑣗𑣆 𑣘𑣆𑣒𑣂 𑣜𑣈 𑣔𑣈𑣊𑣋𑣁
𑣈𑣖 𑣞𑣓𑣁𑣊 𑣕𑣓 𑣌𑣉 𑣓𑣈𑣓 𑣜𑣈 𑣗𑣃 𑣉𑣕𑣁𑣅𑣁. 𑢹𑣉𑣉 𑣎𑣋𑣁𑣜 𑢹𑣗𑣁𑣜𑣊 𑢯𑣂𑣕𑣂 𑣓𑣂𑣕𑣂𑣜𑣈 𑣚𑣋𑣂𑣑
𑣕𑣂𑣞𑣂𑣊𑣄 𑣔𑣂𑣓 𑣜𑣈 𑣔𑣉𑣜𑣌𑣜𑣁 𑣏𑣂 𑣉𑣓𑣚𑣁𑣆𑣓 𑣘𑣚𑣈𑣈𑣒𑣘𑣉𑣉𑣜𑣖 𑣜𑣈 𑣚𑣁𑣄 𑣕𑣈 𑣚𑣁𑣄 𑣗𑣃
𑣓𑣂𑣕𑣂𑣜𑣈𑣅𑣁
𑢹𑣉𑣉 𑢿𑣂𑣌𑣂𑣘𑣂𑣂𑣑𑣂𑣅𑣁 𑣗𑣆 𑣜𑣈 𑣔𑣈𑣊𑣋𑣁 𑣈𑣖 𑣕𑣓𑣂𑣄 𑢶𑣊𑣋𑣃 𑢸𑣃𑣜𑣕𑣂 𑢿𑣂𑣌𑣂𑣘𑣂𑣂𑣑𑣂𑣅𑣁 𑣜𑣈
𑣏𑣂𑣚𑣂𑣌𑣈 𑣕𑣈 𑣔𑣈𑣊𑣋𑣁 𑣈𑣖 𑣔𑣆𑣅𑣉𑣄 𑣖𑣈𑣓𑣈 𑣕𑣈𑣅𑣄 𑣌𑣎𑣂 𑣖𑣜𑣞𑣁𑣚 𑣚𑣋𑣂𑣑 𑣟𑣂𑣑𑣂𑣅𑣉 𑣌𑣉 𑣌𑣂𑣓
𑣗𑣆 𑣕𑣑𑣁
𑢹𑣂𑣕𑣑 : 𑢿𑣂𑣌𑣂𑣖𑣂𑣂𑣑𑣂𑣅𑣁 𑢦𑣓𑣌𑣅𑣃𑣗𑣈𑣈𑣒𑣜 𑣜𑣈 𑣁𑣌𑣁𑣇𑣀𑣒 𑣗𑣆
𑢹𑣂𑣕𑣑 : 𑢾𑣈𑣒𑣂𑣊 𑣌𑣉 𑣉𑣀𑣑𑣉𑣄 𑢱𑣂𑣞𑣘𑣚𑣈𑣈 𑣘𑣉𑣉𑣀𑣒 𑣑𑣁𑣇𑣓𑣚𑣉𑣉𑣑
𑢹𑣂𑣕𑣑 : 𑢦𑣓𑣘𑣃𑣒 𑢹𑣗𑣁𑣜𑣊 𑢯𑣂𑣕𑣂 𑣕𑣈 𑣉𑣚 𑣉𑣀𑣑𑣉𑣄 𑢨𑣑𑣂𑣒𑣂𑣊
𑢡𑣚𑣈𑣈 𑣉𑣉 𑣖𑣂𑣈𑣑 𑣟𑣂𑣑𑣂𑣅𑣉 𑣚𑣈 𑣗𑣆 𑣚𑣈𑣑 𑣕𑣆𑣌𑣈𑣓𑣁 𑣆𑣀𑣒𑣜𑣓𑣈𑣒 𑣜𑣈 𑢥𑣃𑣓𑣂𑣌𑣉𑣉𑣑 𑣕𑣈
