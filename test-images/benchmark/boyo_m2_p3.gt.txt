𑢺𑣈𑣌𑣁-𑣓𑣈𑣚 𑢹𑣉𑣉 𑢧𑣆𑣌𑣂𑣘𑣂𑣑𑣂𑣁 𑣜𑣈 𑢮𑣉𑣟𑣁𑣜!
𑢳𑣈𑣓𑣁 𑣔𑣉 𑢹𑣉𑣉 𑣎𑣋𑣜 𑢹𑢥𑢶 𑣕𑣈 𑣗𑣆𑣅𑣉𑣄 𑣕𑣓 '''𑢹𑢩𑢩 𑢧𑢦𑢬𑢢𑢸𑢢𑢱𑢢𑢡''' 𑣕𑣓𑣁. 𑢳𑣈𑣓𑣁
'''𑢷𑢺𑢨𑢤 𑢧𑢦𑢬𑢢𑢸𑢢𑢱𑢢𑢡''' 𑣕𑣓𑣁 𑣁𑣇𑣜𑣂 𑣗𑣆 𑣞𑣈𑣌𑣁𑣓𑣁
𑢳𑣈𑣓𑣁 𑣞𑣜𑣂 (𑣞𑣃𑣓𑣈) 𑢹𑣉𑣉 𑢧𑣆𑣌𑣂𑣘𑣂𑣑𑣂𑣁 𑣔𑣉 𑣌𑣁 𑣕𑣓𑣁. 𑢳𑣈𑣕𑣁𑣄 𑣜𑣈 𑣚𑣚𑣂-𑣞𑣀𑣋𑣂 𑢩𑣓𑣉𑣚𑣚
𑣌𑣉 𑣉𑣚𑣈 𑣈𑣓𑣀 𑣞𑣜𑣂 (𑢾𑣃𑣓𑣈) '''𑢶𑢼𑢠 𑢹𑢩𑢩 𑢧𑢦𑢬𑢢𑢸𑢢𑢱𑢢𑢡''' 𑣔𑣉 𑣗𑣆𑣅𑣉𑣄𑣁. 𑢡𑣗𑣈𑣓𑣉 𑣉𑣉
𑢩𑣓𑣉𑣚𑣚 𑣌𑣉 𑣉𑣚 𑣔𑣈𑣊𑣋𑣁𑣚𑣈 𑣗𑣈𑣓
𑢷𑣉𑣔𑣜𑣁
''𑢹𑣉𑣉 𑣎𑣋𑣜 𑣁𑣔𑣁𑣓 𑣌𑣉 𑣖𑣜 𑣁𑣜𑣒𑣂𑣌𑣚 (𑢩𑣓𑣉𑣚𑣚) 𑣌𑣉 𑣗𑣃 𑣉𑣚 𑣔𑣈𑣘𑣈𑣊𑣋𑣄 𑣁𑣜 𑣗𑣃 𑣖𑣃𑣚𑣂
𑣔𑣈𑣊𑣋𑣄.''
'''𑢳𑣈𑣓 𑣜𑣈 𑣓𑣖𑣁 𑢩𑣓𑣉𑣚𑣚 𑣜𑣈𑣅𑣄 𑣓𑣃𑣕𑣃𑣖 𑣉𑣚𑣈 𑣗𑣈𑣓 '''
𑢳𑣖𑣁 𑢩𑣓𑣉𑣚𑣚 𑣗𑣆 𑣓𑣀 𑣓𑣈𑣓 𑣜𑣈 𑣉𑣕𑣁𑣅 𑣗𑣈𑣓
𑢡𑣜𑣂𑣌𑣚 (𑢩𑣓𑣉𑣚𑣚) 𑢳𑣃𑣕𑣃𑣖 𑣘𑣈𑣜𑣈𑣄 𑣌𑣈𑣙 𑣕𑣈 𑣓𑣈𑣓 𑣜𑣈 𑣉𑣕𑣁 𑣌𑣈𑣙 𑣜𑣈 𑣓𑣖𑣁 𑣁𑣜𑣒𑣂𑣌𑣚
(𑢩𑣓𑣉𑣚𑣚) 𑣉𑣚 𑣔𑣆𑣅𑣉𑣄𑣁𑣁
𑢳𑣈𑣓𑣕𑣁𑣄 𑣜𑣈 𑣉𑣚𑣁 𑣌𑣓𑣁𑣄 𑣌𑣚𑣂𑣊 𑣗𑣃𑣎𑣁𑣉 (𑣞𑣉𑣖𑣎𑣁𑣉) 𑣕𑣓𑣁
𑢩𑣚𑣁𑣌𑣓 𑢩𑣓𑣉𑣚𑣚 𑣌𑣉:
𑢹𑣉𑣉 𑢷𑢡𑢬𑢰𑢡 (𑢹𑣗𑣜𑣁𑣊 𑢯𑣂𑣕𑣂) 𑣕𑣈
𑢯𑣁𑣜𑣂 𑢷𑣃𑣜𑣃 𑢴𑣃𑣜𑣈𑣊𑣗𑣋𑣓
"𑢷𑣂𑣕𑣂𑣅𑣈𑣖"𑢴𑣃𑣓𑣃𑣟𑣈 𑢮𑣉𑣙𑣁𑣜
𑢮𑣅𑣜𑣁𑣌𑣁𑣀𑣑 𑣜𑣈𑣓 𑢹𑣞𑣃𑣜 𑢾𑣂𑣍𑣙𑣞𑣁, 𑢵𑣃𑣜 𑢾𑣂𑣍𑣙𑣞𑣁 𑣉𑣀𑣑𑣉𑣄 𑢩𑣛𑣂𑣞𑣈 𑣜𑣈𑣓 𑢬𑣂𑣍𑣎𑣜𑣂,
𑣙𑣋𑣁-𑣎𑣋𑣁𑣜 𑣕𑣓𑣁
𑢾𑣗𑣈𑣓 𑣈𑣕𑣈 𑣚𑣚𑣂 𑢹𑣉𑣉 𑣎𑣋𑣁𑣜 𑣔𑣉 𑢮𑣅𑣜𑣁𑣌𑣁𑣀𑣑 𑣜𑣈𑣓 𑢹𑣞𑣃𑣜 𑢾𑣂𑣍𑣗𑣃𑣆,
𑢹𑣗𑣜𑣁𑣊 𑢯𑣂𑣕𑣂 𑣔𑣉 𑢹𑣉𑣉 𑣎𑣋𑣁𑣜 𑣉𑣚 𑣘𑣆𑣒𑣂 𑣜𑣈 𑣙𑣃𑣎𑣃𑣄 𑣕𑣓 𑣚𑣂𑣘𑣂 𑣕𑣓𑣁. 𑢳𑣈𑣓 𑣚𑣂𑣘𑣂 𑣔𑣉
𑢺𑣌𑣉 𑢷𑣉𑣔𑣜𑣁 𑣕𑣂𑣌𑣂 𑣘𑣍𑣎𑣁 𑣓𑣖 𑣚𑣈𑣑 𑣕𑣆𑣌𑣈𑣓𑣁. 𑢳𑣈𑣓 𑣚𑣂𑣘𑣂 𑣜𑣈 𑣏𑣂𑣕𑣂 𑣌𑣉 𑣖𑣈𑣓𑣄. 𑢳𑣈𑣓
𑣏𑣂𑣕𑣂 𑣌𑣉 𑣜𑣈 𑣔𑣉 𑣆𑣘𑣁𑣓 𑣗𑣜𑣉𑣊 𑣌𑣉, 𑣔𑣉 𑣗𑣈𑣊𑣓 𑣗𑣜𑣉𑣊 𑣉𑣀𑣑𑣉𑣄 𑣔𑣉 𑣖𑣃𑣇𑣑 𑣏𑣂𑣕𑣂
'𑢵' 𑢫𑣃𑣜𑣃, 𑢬𑣉𑣚 𑢺𑣌𑣉 𑢷𑣉𑣔𑣜𑣁
