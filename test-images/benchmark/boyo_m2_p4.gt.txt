𑢹𑣗𑣁𑣜𑣊 𑢯𑣂𑣕𑣂 𑢦𑣓𑣂𑣕𑣃
''𑢵𑣂𑣂𑣎𑣈𑣓'' 𑣔𑣉 𑣜𑣈𑣑𑣂𑣅𑣉 𑣏𑣈𑣓𑣈𑣚 𑣘𑣅𑣁𑣜 𑣌𑣈𑣈 𑣞𑣓𑣂𑣎𑣜 𑣔𑣆 𑣆𑣔𑣂𑣅𑣃𑣜-𑣔𑣅𑣓 𑣎𑣓𑣄 𑣕𑣓𑣁.
𑢷𑣂𑣜𑣞𑣁 𑢶𑣃𑣀𑣑𑣁𑢷𑣂𑣜𑣞𑣁 𑢶𑣃𑣓𑣑𑣁
𑢷𑣂𑣜𑣞𑣁 𑢶𑣃𑣀𑣑𑣁 ( 𑢳𑣉𑣈𑣖𑣗𑣁𑣜 𑣈𑣕𑣈 𑢮𑣃𑣃𑣓 ) 𑣔𑣉 𑢹𑣜𑣕𑣙𑣞𑣁 𑣜𑣈𑣓 𑣁𑣔𑣂𑣗𑣁𑣞𑣂 𑣞𑣟𑣁𑣕𑣀𑣕𑣜𑣁
𑣞𑣀𑣋𑣜𑣁𑣖𑣂𑣎 𑣕𑣆𑣌𑣈𑣓𑣁
𑢷𑣂𑣜𑣞𑣁 𑢶𑣃𑣀𑣑𑣁 𑣔𑣉 𑢷𑣊𑣋𑣁𑣚 𑢸𑣜𑣈𑣞𑣂𑣑𑣈𑣀𑣞𑣂 𑣜𑣈𑣅𑣄 𑢧𑣚𑣂𑣙𑣕𑣃 𑣜𑣈𑣎 𑣎𑣉𑣓𑣉𑣖 𑣚𑣈𑣓𑣁, 𑣉𑣌𑣉𑣓𑣁
𑣏𑣂 𑣕𑣂𑣞𑣂𑣊 𑣔𑣉 𑢮𑣅𑣜𑣁𑣌𑣁𑣀𑣑 𑣜𑣈𑣓 𑢬𑣃𑣃𑣀𑣒𑣂 𑣜𑣈 𑣖𑣈𑣓𑣄. 𑢴𑣃𑣜𑣈𑣊 𑣌𑣉 𑣜𑣈 𑣔𑣉 𑣁𑣎𑣄 𑣙𑣕𑣃
𑢧𑣚𑣂𑣙𑣕𑣃 𑣉𑣉 𑣖𑣈𑣓𑣉𑣄𑣁 𑣉𑣀𑣑𑣉𑣄 𑢯𑣚𑣌𑣁𑣑 𑣉𑣉 𑣖𑣈𑣓𑣉𑣄𑣁. 𑢧𑣚𑣂𑣙𑣕𑣃 𑣔𑣉 𑣁𑣘𑣃𑣕𑣈 𑢾𑣃𑣋𑣓𑣁
𑢶𑣃𑣀𑣑𑣁 𑣁𑣄 𑣙𑣕𑣃 𑣕𑣆𑣌𑣈𑣓𑣁
𑢷𑣂𑣜𑣞𑣂𑣊 𑢮𑣁𑣖𑣃𑣔𑣁,𑢦𑣏𑣄𑣘𑣂𑣙
𑢷𑣂𑣆𑣜 𑣜𑣈 𑣈𑣞𑣌𑣜 𑣕𑣆𑣓 𑣕𑣓𑣂𑣄 𑣔𑣉 𑣗𑣂𑣜 𑣌𑣃𑣚𑣚𑣈 𑣌𑣉 𑣖𑣈𑣕𑣁𑣆𑣅𑣈
𑢦𑣓𑣂𑣙 𑣋𑣈 𑣓𑣈𑣓 𑢩𑣓𑣉𑣚𑣚 𑣜𑣈𑣄 𑣕𑣜𑣞𑣚𑣉𑣄 𑣕𑣓𑣁,𑢦𑣓𑣂𑣙 𑣔𑣉𑣙𑣉𑣉𑣌𑣉 𑣖𑣈𑣕𑣁𑣆𑣅𑣈 "𑢷𑣂𑣜𑣞𑣂𑣊"
𑢸𑣉𑣒𑣉 𑢹𑣉𑣉, 𑣞𑣜𑣂 𑣓𑣂𑣕𑣃𑣖 𑢬𑣛𑣈𑣅𑣁 𑢱𑣂𑣗𑣈𑣜 𑢷𑣁𑣀𑣑𑣜𑣁, 𑣔𑣉 𑣜𑣈𑣓 𑢬𑣉𑣉𑣚 𑣟𑣂𑣔𑣜𑣉𑣙 𑣜𑣈𑣓
𑣗𑣂𑣟𑣜 𑣕𑣓𑣁𑣎. 𑢡𑣎 𑢳𑣉𑣟𑣁𑣖𑣗𑣁𑣜 𑣜𑣈𑣅𑣄 𑢾𑣈𑣜𑣈𑣊𑣞𑣂𑣅𑣈 𑣚𑣛𑣆𑣎 𑣁𑣅𑣁𑣜 𑣚𑣈𑣑 𑣕𑣆𑣌𑣈𑣓𑣁. 𑢡𑣅𑣁𑣄
𑣙𑣕𑣃 𑣔𑣉 𑢼𑣎𑣁𑣗𑣞𑣁 𑣕𑣆𑣌𑣈𑣓𑣁. 𑢿𑣂𑣑𑣜𑣉𑣙 𑣌𑣈𑣑 𑣚𑣋𑣂𑣑 𑢷𑣜𑣂𑣒𑣂𑣞 𑣞𑣜𑣌𑣁𑣜 𑣁𑣎 𑣓𑣖 𑣚𑣋𑣂𑣑
𑣋𑣉𑣒𑣁 𑣖𑣃𑣚𑣂 𑣁𑣅𑣄 𑣘𑣉𑣒𑣉 𑣌𑣉 𑣞𑣒𑣁𑣉 𑣚𑣈𑣑𑣁, 𑣈𑣓𑣁 𑣕𑣈 𑣁𑣎 𑣔𑣉 𑣙𑣉𑣉 𑣌𑣉 𑣕𑣚𑣁 𑣜𑣈 𑢸𑣉𑣒𑣉
𑢹𑣉𑣉 𑣖𑣈𑣓𑣕𑣈𑣎 𑣓𑣃𑣕𑣃𑣖 𑣎𑣓𑣁. 𑢵𑣅𑣉𑣖 𑣔𑣉 𑣎𑣓𑣟𑣁𑣜𑣂 𑣜𑣈 𑣌𑣉 𑢳𑣜𑣁 𑢹𑣉𑣉, 𑢱𑣈𑣗𑣁𑣅 𑢹𑣉𑣉
𑣉𑣀𑣑𑣉𑣄 𑢷𑣉𑣛𑣉 𑢹𑣉𑣉 𑣚𑣉𑣄 𑣌𑣉 𑣘𑣞𑣈 𑣋𑣉𑣄𑣈 𑣌𑣂𑣎𑣈
