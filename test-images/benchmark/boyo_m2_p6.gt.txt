'''𑢳𑣂𑣙𑣉𑣓''' (𑢳𑣂𑣙𑣉𑣊𑣎𑣜:), 𑣓𑣈𑣓𑣁 𑣕𑣃𑣜 𑢨𑣞𑣂𑣅𑣈 𑢹𑣘𑣁-𑣕𑣓-𑣔𑣄 𑣜𑣈 𑣖𑣈𑣓𑣄 𑣕𑣓 𑣖𑣂𑣅𑣑
𑣔𑣄-𑣑𑣂𑣘𑣈 𑣔𑣂𑣞𑣃𑣖 𑣕𑣓𑣁. 𑢳𑣈𑣓 𑣔𑣂𑣞𑣃𑣖 𑣙𑣞𑣃𑣜 𑣞𑣄 𑣜𑣈 𑢳𑣂𑣙𑣉𑣓-𑣔𑣄, 𑢵𑣃𑣜 𑢾𑣂𑣅𑣈𑣓-𑣔𑣄,
𑣙𑣉 𑣓𑣃𑣕𑣃𑣖𑣁𑣌𑣓𑣁
𑢴𑣂𑣅𑣃𑣜𑣂 𑣔𑣉 𑢹𑣉𑣉 𑣌𑣉 𑣙𑣕𑣃 𑣜𑣈 𑣗𑣉𑣊𑣋𑣁 𑣗𑣃𑣜𑣃 𑣘𑣆𑣒𑣂 𑣌𑣉 𑣞𑣗 𑣕𑣓𑣂𑣎. 𑢳𑣈𑣚 𑣓𑣖𑣉𑣄𑣁 𑣏𑣂
𑣌𑣉𑣁 𑣌𑣉 𑣋𑣈 𑣔𑣂𑣅𑣃𑣜𑣂𑣓𑣈. 𑢹𑣕𑣃 𑣜𑣈𑣅𑣄 𑣎𑣈𑣕𑣁𑣓 𑣘𑣉𑣜𑣉𑣗 𑣘𑣃𑣓𑣆 𑣌𑣉 𑣜𑣈 𑣔𑣂𑣅𑣃𑣜𑣂 𑣋𑣈
𑣗𑣉𑣊𑣋𑣁𑣅𑣁. 𑢾𑣗𑣈𑣓 𑣌𑣂𑣚𑣂 𑣜𑣈𑣓 𑢹𑣉𑣉 𑣌𑣉 𑣔𑣂𑣅𑣃𑣜𑣂 𑣔𑣆𑣓𑣈
𑢴𑣂𑣅𑣃𑣜𑣂 𑣔𑣉 𑣘𑣉𑣜𑣉𑣗 𑣌𑣉 𑣜𑣈 𑣗𑣉𑣊𑣋𑣁𑣅 𑣕𑣈𑣅𑣄 𑣋𑣈 𑣖𑣜𑣁𑣊 𑣘𑣆𑣒𑣂 𑣔𑣉
