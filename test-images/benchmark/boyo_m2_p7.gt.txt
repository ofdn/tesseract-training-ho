𑣕𑣓𑣁. 𑢳𑣈𑣓 𑣘𑣉𑣜𑣉𑣗 𑣔𑣉 𑣙𑣈𑣜 𑣕𑣑 𑣗𑣗𑣁 𑣖𑣕𑣁 𑣕𑣓 𑣜𑣈 𑣘𑣉𑣜𑣉𑣗𑣉𑣄𑣁. 𑢹𑣈𑣜𑣉𑣄 𑣇𑣜𑣂 𑣗𑣉𑣊𑣋𑣁𑣅
𑣜𑣈 𑣋𑣈 𑣎𑣁 𑣚𑣈𑣌𑣁𑣓 𑣓𑣖𑣁 𑣎𑣉𑣉 𑣑𑣁𑣚𑣂 𑣌𑣉 𑣌𑣁 𑣎𑣉𑣖𑣉𑣄𑣁

𑢴𑣈𑣞𑣁𑣇𑣚𑣂 𑣔𑣉 𑢹𑣉𑣉 𑣌𑣉 𑣁𑣄 𑣙𑣕𑣃 𑣗𑣉𑣊𑣋𑣁 𑣕𑣓𑣁𑣎. 𑢾𑣗𑣈𑣓 𑢹𑣉𑣉 𑣙𑣕𑣃 𑣌𑣉 𑣖𑣂𑣈𑣑 𑣌𑣃𑣒𑣂 𑣜𑣈
𑣔𑣜𑣃 𑣌𑣉 𑣎𑣃𑣖𑣗𑣈𑣈 𑣕𑣆𑣓𑣈, 𑣈𑣓𑣁 𑣋𑣈 𑢴𑣈𑣞𑣁𑣇𑣚𑣂 𑣁𑣄 𑣗𑣞𑣁 𑣖𑣈𑣓𑣕𑣈 𑣖𑣈𑣓𑣉𑣄𑣁. 𑢴𑣈𑣞𑣁𑣇𑣚𑣂

𑢵𑣂𑣖𑣘𑣃
𑢷𑣈𑣚𑣎𑣂𑣅𑣖 𑢥𑣃𑣃𑣜𑣉𑣗 𑣜𑣈 𑣖𑣈𑣓𑣄 𑣕𑣓 𑣖𑣂𑣅𑣑 𑣔𑣂𑣞𑣃𑣖 𑣕𑣓𑣁. 𑢳𑣈𑣓𑣁 𑣜𑣈𑣅𑣄 𑣁𑣜𑣞𑣆 𑢷𑣜𑣃𑣞𑣈𑣚
𑢷𑣉𑣉𑣕𑣞𑣟𑣓𑣁 𑢡𑣘𑣂𑣜𑣌𑣈 𑣜𑣈 𑣖𑣈𑣓𑣄 𑣕𑣓 𑣖𑣂𑣅𑣑 𑣔𑣂𑣞𑣃𑣖 𑣕𑣓𑣁. 𑢳𑣈𑣓𑣁 𑣜𑣈𑣅𑣄 𑣁𑣜𑣞𑣆 𑢫𑣗𑣉𑣜𑣉𑣓𑣈
