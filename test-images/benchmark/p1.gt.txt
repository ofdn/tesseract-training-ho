𑢳𑣈𑣓𑣁 𑣔𑣉 𑢹𑣉𑣉 𑣎𑣋𑣜 𑢹𑢥𑢶 𑣕𑣈 𑣗𑣆𑣅𑣉𑣄 𑣕𑣓 𑢹𑢩𑢩 𑢧𑢦𑢬𑢢𑢸𑢢𑢱𑢢𑢡 𑣕𑣓𑣁. 𑢳𑣈𑣓𑣁 𑢷𑢺𑢨𑢤
𑢧𑢦𑢬𑢢𑢸𑢢𑢱𑢢𑢡 𑣕𑣓𑣁 𑣁𑣇𑣜𑣂 𑣗𑣆 𑣞𑣈𑣌𑣁𑣓𑣁. 𑢳𑣈𑣓𑣁 𑣞𑣜𑣂 (𑣞𑣃𑣓𑣈) 𑢹𑣉𑣉 𑢧𑣆𑣌𑣂𑣘𑣂𑣑𑣂𑣁 𑣔𑣉 𑣌𑣁
𑣕𑣓𑣁. 𑢳𑣈𑣕𑣁𑣄 𑣜𑣈 𑣚𑣚𑣂-𑣞𑣀𑣋𑣂 𑢩𑣓𑣉𑣚𑣚 𑣌𑣉 𑣉𑣚𑣈 𑣈𑣓𑣀 𑣞𑣜𑣂 (𑢾𑣃𑣓𑣈) 𑢶𑢼𑢠 𑢹𑢩𑢩
𑢧𑢦𑢬𑢢𑢸𑢢𑢱𑢢𑢡 𑣔𑣉 𑣗𑣆𑣅𑣉𑣄𑣁. 𑢡𑣗𑣈𑣓𑣉 𑣉𑣉 𑢩𑣓𑣉𑣚𑣚 𑣌𑣉 𑣉𑣚 𑣔𑣈𑣊𑣋𑣁𑣚𑣈 𑣗𑣈𑣓
𑢳𑣈𑣓𑣁 𑣗𑣚𑣈𑣄 𑢧𑣆𑣌𑣂𑣘𑣂𑣑𑣂𑣁 𑣕𑣓𑣁, 𑣈𑣓𑣁 𑣖𑣈𑣓𑣕𑣈 𑣓𑣈𑣙𑣕𑣄 𑣜𑣈 𑣉𑣚𑣉𑣄 𑣕𑣓 𑢩𑣓𑣉𑣚𑣚 𑣌𑣉
𑣓𑣈𑣚𑣈𑣌𑣁 𑣓𑣃𑣕𑣃𑣖 𑣕𑣆𑣓 𑣚𑣋𑣁𑣕𑣂𑣊𑣈: 𑢩𑣓𑣉𑣚𑣚 𑣜𑣈𑣅𑣄 𑣓𑣃𑣕𑣃𑣖. 𑢮𑣁𑣁 𑣖𑣈𑣓𑣕𑣈 𑢺𑣌𑣉 𑢷𑣉𑣔𑣜𑣁
𑣜𑣈𑣅𑣄 𑣖𑣂𑣅𑣈𑣑 𑣁𑣜𑣒𑣂𑣌𑣚 𑣗𑣈𑣓 𑣉𑣚𑣈 𑣕𑣓𑣁, 𑣈𑣓𑣔𑣉 𑣈𑣓𑣁 𑣓𑣈𑣚𑣈𑣌𑣁 𑣓𑣃𑣕𑣃𑣖𑣈𑣗𑣈𑣓: 𑢺𑣌𑣉
𑢹𑣉𑣉 𑣎𑣋𑣜 𑣁𑣔𑣁𑣓 𑣌𑣉 𑣖𑣜 𑣁𑣜𑣒𑣂𑣌𑣚 (𑢩𑣓𑣉𑣚𑣚) 𑣌𑣉 𑣗𑣃 𑣉𑣚 𑣔𑣈𑣘𑣈𑣊𑣋𑣄 𑣁𑣜 𑣗𑣃 𑣖𑣃𑣚𑣂
𑢳𑣈𑣓𑣕𑣄 𑣜𑣈 𑣓𑣁𑣄 𑣎𑣉𑣊 𑣜𑣈 𑢩𑣓𑣉𑣚𑣚 𑣌𑣉𑣋𑣈 𑣖𑣈𑣓𑣄'𑣁
𑢺𑣕𑣜 𑣜𑣈 𑣈𑣖𑣁 𑣌𑣓 𑣑𑣚𑣁 (𑣑𑣗𑣁) 𑣜𑣈 𑣉𑣕𑣁 𑣌𑣈𑣙 𑣕𑣈 𑣓𑣖𑣁 𑣉𑣓𑣉𑣚𑣚 𑣜𑣈𑣅𑣄 𑣓𑣃𑣕𑣃𑣖 𑣉𑣚 𑣌𑣈𑣙
𑣕𑣈 𑢳𑢶𑢡 𑢩𑣓𑣉𑣚𑣚 𑣗𑣆 𑣗𑣈𑣓
𑢮𑣋𑣜-𑢮𑣓𑣁𑣋𑣜 𑣔𑣉 𑢹𑣜𑣕𑣙𑣞𑣁 𑣜𑣈 𑣪 𑣑𑣙𑣌𑣁 𑣙𑣉𑣛𑣉𑣌𑣉 𑣎𑣋𑣁𑣜𑣈 𑣕𑣓 𑣉𑣉𑣞𑣒𑣜𑣉-𑣈𑣞𑣂𑣁𑣒𑣂𑣌
𑣋𑣉𑣉𑣒𑣉 𑣜𑣈𑣓 𑢶𑣃𑣀𑣑𑣁 𑣎𑣋𑣁𑣜 𑣕𑣓𑣁. 𑢳𑣈𑣓𑣁 𑢹𑣉𑣉 𑣎𑣕𑣂 𑣌𑣉 𑣎𑣋𑣁𑣜𑣈 𑣙𑣅𑣖 𑣕𑣓𑣁 𑣉𑣀𑣑𑣉𑣄𑣄
𑢹𑣗𑣜𑣁𑣊 𑢯𑣂𑣕𑣂 𑣕𑣈 𑣉𑣚𑣉𑣄𑣁. 𑢩𑣌𑣉𑣓𑣉𑣌𑣉𑣓 𑣕𑣄 𑣌𑣉 𑣜𑣈 𑢴𑣈𑣉𑣓𑣁𑣋𑣜𑣂, 𑢩𑣛𑣂𑣅𑣁, 𑣉𑣀𑣑𑣉𑣄𑣄
𑢺𑣅𑣒𑣂𑣓 𑣚𑣂𑣘𑣂 𑣘𑣆𑣒𑣂 𑣜𑣈 𑣁𑣋𑣃𑣃𑣄 𑣕𑣓𑣁, 𑣖𑣈𑣓𑣔𑣉 𑢹𑣉𑣉 𑣎𑣋𑣁𑣜𑣈 𑣕𑣓 𑣌𑣉 𑣖𑣈𑣓𑣈𑣅𑣁 𑣏𑣂
𑣈𑣓𑣁𑣌𑣉 𑢹𑣉𑣉 𑣎𑣋𑣁𑣜 𑣚𑣋𑣂𑣑 𑣌𑣁 𑣗𑣃𑣋𑣂𑣓𑣈, 𑢹𑣗𑣜𑣁𑣊 𑢯𑣂𑣕𑣂 𑣋𑣈. 𑢸𑣃𑣜𑣄 𑣞𑣊𑣋𑣂 𑢹𑣉𑣉 𑣎𑣋𑣁𑣜
𑢮𑣅𑣜𑣁𑣌𑣁𑣀𑣑 𑣜𑣈𑣓 𑢹𑣞𑣃𑣜 𑢾𑣂𑣍𑣙𑣞𑣁, 𑢵𑣃𑣜 𑢾𑣂𑣍𑣙𑣞𑣁 𑣉𑣀𑣑𑣉𑣄 𑢩𑣛𑣂𑣞𑣈 𑣜𑣈𑣓 𑢬𑣂𑣍𑣎𑣜𑣂
𑢶𑣉𑣜𑣗𑣉𑣉𑣍𑣎𑣉 𑣜𑣈 𑣎𑣋𑣁𑣜𑣉𑣄 𑣕𑣓𑣁. 𑢹𑣉𑣉 𑣔𑣉 𑢶𑣃𑣀𑣑𑣁𑣜𑣂 𑣉𑣀𑣑𑣉𑣄𑣄 𑢾𑣀𑣕𑣁𑣚𑣂 𑣌𑣉 𑣜𑣈𑣅𑣄
𑢾𑣗𑣈𑣓 𑣈𑣕𑣈 𑣚𑣚𑣂 𑢹𑣉𑣉 𑣎𑣋𑣁𑣜 𑣔𑣉 𑢮𑣅𑣜𑣁𑣌𑣁𑣀𑣑 𑣜𑣈𑣓 𑢹𑣞𑣃𑣜 𑢾𑣂𑣍𑣗𑣃𑣆
𑢾𑣜𑣆𑣌𑣈𑣚𑣁-𑢬𑣉𑣜𑣞𑣉𑣁𑣀, 𑢵𑣃𑣜 𑢾𑣂𑣍𑣗𑣃𑣆 𑣜𑣈 𑢩𑣛𑣂𑣞𑣁 𑣜𑣈𑣓 𑢶𑣉𑣜𑣗𑣉𑣉𑣍𑣎𑣉, 𑢬𑣈𑣅𑣉𑣍𑣎𑣉𑣜 𑣎𑣂𑣚𑣁
𑣌𑣉 𑣜𑣈 𑣎𑣋𑣁𑣜𑣉𑣄 𑣕𑣓𑣁. 𑢹𑣉𑣉 𑣎𑣋𑣁𑣜 𑣔𑣉 𑢾𑣀𑣕𑣁𑣚𑣂, 𑢶𑣃𑣀𑣑𑣁𑣜𑣂 𑣉𑣀𑣑𑣉𑣄 𑢷𑣃𑣖𑣂𑣎 𑣎𑣋𑣁𑣜 𑣌𑣉
𑣚𑣉𑣄 𑣖𑣂𑣑 𑣋𑣉𑣉𑣒𑣉 𑣜𑣈 𑣋𑣈 𑣙𑣒𑣂𑣊 𑣁𑣌𑣁𑣓𑣁. 𑢹𑣉𑣉 𑣎𑣋𑣁𑣜 𑣉𑣀𑣑𑣉𑣄 𑢶𑣃𑣀𑣑𑣁𑣜𑣂 𑣎𑣋𑣁𑣜 𑣔𑣉
𑣖𑣂𑣑 𑣚𑣈𑣌𑣁𑣓 𑣎𑣋𑣁𑣜 𑣌𑣉 𑣕𑣓𑣁, 𑣚𑣂𑣊𑣋𑣃𑣆𑣞𑣒 𑣌𑣉 𑣖𑣈𑣓𑣈𑣅𑣁 𑣏𑣂 𑢹𑣉𑣉 𑣎𑣋𑣁𑣜 𑣉𑣀𑣑𑣉𑣄
𑢶𑣃𑣀𑣑𑣁𑣜𑣂 𑣎𑣋𑣁𑣜 𑣔𑣉 𑣖𑣂𑣑 𑣋𑣈𑣅𑣁
𑢹𑣉𑣉 𑣎𑣋𑣁𑣜 𑣔𑣉 𑢹𑣗𑣁𑣜𑣊 𑢯𑣂𑣕𑣂 𑣚𑣂𑣘𑣂 𑣕𑣈 𑣉𑣚𑣉𑣄 𑣕𑣓𑣁
𑢷𑣁𑣌𑣐𑣁 𑣏𑣈𑣓𑣄 𑣌𑣉 𑣖𑣈𑣕𑣄?
𑢩𑣌𑣉𑣓 𑣗𑣂𑣑𑣔𑣁 𑣕𑣈 𑣓𑣂𑣜𑣚𑣁 𑣉𑣓𑣉𑣚, 𑣘𑣛𑣁𑣉 𑣁𑣜 𑣎𑣋𑣜 𑣞𑣉𑣅𑣕𑣁 𑣈𑣕𑣉𑣄 𑣁𑣄 𑣈𑣓 𑣗𑣂𑣑𑣔𑣁 𑣋𑣈𑣌𑣉
