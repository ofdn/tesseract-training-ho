𑢯𑣂𑣚𑣂𑣌𑣁:- 𑢹𑣁𑣜𑣁-𑢹𑣁𑣁𑣜𑣁, 𑢼𑣂𑣕𑣁-𑢼𑣂𑣂𑣕𑣁, 𑢺𑣂𑣒𑣁-𑢺𑣂𑣂𑣒𑣁, 𑢼𑣈𑣕𑣁-𑢼𑣈𑣈𑣕𑣁 𑣈𑣖𑣓𑣄
𑢷𑣈𑣊𑣓 𑣎𑣈𑣜𑣖𑣈𑣙𑣑 𑣏𑣈𑣓𑣄 𑣌𑣉 𑣖𑣈𑣕𑣄?
𑢷𑣈𑣊𑣓 𑣗𑣜𑣊 𑣌𑣉 𑣎𑣈𑣜𑣜𑣈𑣓 𑣜𑣈 𑣉𑣌𑣉𑣓 𑣜𑣃𑣃𑣘 𑣗𑣉𑣔𑣉𑣚𑣈𑣓𑣁 𑣈𑣓𑣁 𑣋𑣈𑣌𑣉 𑣖𑣈𑣕𑣄 𑢷𑣈𑣊𑣓
𑢯𑣂𑣚𑣂𑣌𑣁:- 𑢶𑣘𑣘𑣄, 𑢶𑣂𑣚𑣚,𑢽𑣃𑣜𑣜,𑢾𑣃𑣚𑣚 𑣈𑣖𑣁𑣓𑣄
𑢽𑣗𑣔 𑣗𑣂𑣏𑣁𑣜 𑣏𑣈𑣓𑣄 𑣌𑣉 𑣖𑣈𑣕𑣄?
𑢽𑣗𑣔 𑣜𑣈𑣅𑣄 𑣁𑣏𑣁𑣜 𑣗𑣂𑣏𑣁𑣜 𑣗𑣁𑣌𑣐𑣁 𑣜𑣈 𑣈𑣓 𑣘𑣛𑣄 𑣌𑣉 𑣖𑣈𑣕𑣄 𑣉𑣌𑣉𑣓 𑣜𑣈 𑣝𑣗𑣔 𑣌𑣉'𑣄
𑣜𑣉𑣌𑣉𑣖 𑣁𑣔𑣉 𑣌𑣎𑣂 𑣇𑣖𑣂 𑣜𑣃𑣃𑣘 𑣓𑣁 𑣉𑣓𑣑𑣉𑣄 𑣗𑣆 𑣗𑣂𑣏𑣜 𑣙𑣉𑣗𑣁 𑣓𑣁, 𑣈𑣓𑣁 𑣋𑣈𑣌𑣉 𑣖𑣈𑣕𑣄
𑢫𑣉𑣙𑣖 𑣁𑣜 𑢫𑣉𑣙𑣜𑣘𑣉 𑣏𑣈𑣓𑣄 𑣌𑣉 𑣖𑣈𑣕𑣄?
𑢽𑣗𑣔 𑣖𑣃𑣓𑣑𑣂 𑣜𑣈 𑣌𑣜𑣈 𑣔𑣉 𑣝𑣗𑣔 𑣕𑣃𑣓𑣑𑣃 𑣜𑣈 𑣖𑣂𑣘𑣘𑣂𑣅𑣑 𑣝𑣗𑣔 𑣊𑣝 𑣌𑣜𑣈 𑣏𑣂𑣕𑣂 𑣏𑣂 𑣝𑣗𑣔
𑣔𑣉 𑣋𑣉𑣙𑣖 𑣁𑣜 𑣋𑣉𑣙𑣜𑣘𑣉 𑣌𑣉 𑣖𑣈𑣕𑣄
𑢯𑣂𑣚𑣌𑣁:- 𑢷𑣌𑣁𑣀:𑢬𑣋𑣙𑣛𑣂𑣌𑣁 𑣁𑣋𑣃𑣇𑣄. 𑢽𑣗𑣔:𑢬𑣋𑣙𑣛𑣂𑣌𑣁 𑢫𑣉𑣙𑣜𑣘𑣉:𑢬 𑢫𑣉𑣙𑣖:𑢬𑣁 𑢶𑣃𑣇𑣛
𑢽𑣃𑣖 𑣏𑣈𑣓𑣄 𑣌𑣉 𑣖𑣈𑣕𑣄?
𑢩𑣌𑣉𑣓𑣁 𑣈𑣒𑣄 𑣝𑣗𑣔 𑣌𑣉'𑣄 𑣔𑣈𑣊𑣋𑣈 𑣕𑣈 𑣌𑣁 𑣗𑣆𑣓𑣁, 𑣈𑣓𑣁 𑣋𑣈𑣌𑣉 𑣖𑣈𑣕𑣄 𑢽𑣃𑣖
𑢯𑣂𑣚𑣂𑣌𑣁:𑢶𑣈𑣑,𑢶𑣃𑣒𑣁,𑢺𑣃𑣕𑣃𑣜𑣜,𑢵𑣂𑣙,𑢹𑣈𑣓𑣔𑣈,𑢸𑣃𑣓𑣑𑣂,𑣞𑣞𑣀 𑣈𑣖𑣁𑣓𑣄
𑢮𑣓𑣁𑣘𑣄 𑣏𑣈𑣓𑣄 𑣌𑣉 𑣖𑣈𑣕𑣄?
𑢩𑣌𑣉𑣓𑣁 𑣎𑣃𑣛𑣂 𑣚𑣈𑣓𑣅 𑣕𑣈 𑣗𑣆𑣓𑣁 𑣈𑣓𑣌𑣓 𑣝𑣗𑣔 𑣋𑣈𑣌𑣉 𑣖𑣈𑣕𑣄 𑢮𑣓𑣁𑣘𑣄
𑢯𑣂𑣚𑣂𑣌𑣁:- 𑢹𑣁𑣁𑣑 𑣕𑣈𑣅𑣄, 𑢶𑣁𑣀𑣑𑣂 𑣕𑣓𑣂𑣄, 𑢵𑣉𑣟 𑣕𑣇𑣎𑣂𑣄, 𑣁𑣍𑣎𑣙-𑣁𑣍𑣎𑣙, 𑣞𑣞𑣀 𑣀𑣄 𑣈𑣖𑣓
𑢷𑣌𑣁𑣀 𑣜𑣄 𑣘𑣉𑣜𑣕𑣁𑣛 𑣏𑣈𑣓𑣄 𑣌𑣉 𑣖𑣈𑣕𑣄?
𑢷𑣁𑣌𑣐𑣁 𑣜𑣄 𑣜𑣃𑣃𑣚 𑣉𑣛𑣊 𑣚𑣈𑣌𑣁 𑣕𑣈 𑣝𑣗𑣔 𑣖𑣂𑣑 𑣖𑣃𑣚𑣂 𑣘𑣓𑣕𑣂 𑣏𑣂 𑣋𑣘𑣁𑣕𑣂𑣅𑣁 𑣌𑣓𑣈𑣙 𑣕𑣈
